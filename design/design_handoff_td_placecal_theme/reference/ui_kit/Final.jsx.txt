/** Partner event browser: core's per-day lists made visible — pink day headings spanning the two-column grid, a slim toolbar, "Show more" paging, iCal/CSV beneath. */
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const WEEKDAYS = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
const TODAY = new Date(2026, 8, 8);
function expand(events, weeks = 6) {
  // Demo data only lists one date per event; a real partner feed repeats weekly/fortnightly, so project each repeating event forward.
  const out = [];
  events.forEach(e => {
    const first = new Date(2026, MONTHS.indexOf(e.month), Number(e.day));
    const step = /fortnight/i.test(e.repeat || '') ? 14 : /week/i.test(e.repeat || '') ? 7 : /month/i.test(e.repeat || '') ? 28 : 0;
    for (let i = 0, d = new Date(first); i < (step ? weeks * 7 / step : 1); i++, d = new Date(d.getTime() + step * 864e5)) { if (d >= TODAY) out.push({ ...e, dateObj: d, key: e.id + '-' + i }); if (!step) break; }
  });
  return out.sort((a, b) => a.dateObj - b.dateObj || a.time.localeCompare(b.time));
}
const dayLabel = d => { const diff = Math.round((d - TODAY) / 864e5); const base = WEEKDAYS[d.getDay()] + ' ' + d.getDate() + ' ' + MONTHS[d.getMonth()]; return diff === 0 ? 'Today · ' + base : diff === 1 ? 'Tomorrow · ' + base : base; };
function PartnerEvents({ events, navigate }) {
  const [rep, setRep] = React.useState('All'); const [pick, setPick] = React.useState(false); const [date, setDate] = React.useState(''); const [shown, setShown] = React.useState(4);
  const all = React.useMemo(() => expand(events), [events]);
  let list = all.filter(e => rep === 'All' || (rep === 'Regular' ? !!e.repeat : !e.repeat));
  if (date) { const from = new Date(date + 'T00:00'); list = list.filter(e => e.dateObj >= from); }
  const days = []; list.forEach(e => { const k = e.dateObj.toDateString(); const g = days.find(g => g.k === k); g ? g.items.push(e) : days.push({ k, d: e.dateObj, items: [e] }); });
  const visible = days.slice(0, shown); const more = days.length - visible.length;
  if (!events.length) return <><FloatingTitle as="h2" size="0.9rem" style={{ margin: 0 }}>Upcoming partner events</FloatingTitle><EmptyState>There are no listed events for this partner but check back soon!</EmptyState></>;
  return <div style={{ textAlign: 'left', paddingBottom: 40 }}>
    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 16, flexWrap: 'wrap', marginBottom: 8 }}>
      <FloatingTitle as="h2" size="0.9rem" align="left" style={{ margin: 0 }}>Upcoming partner events <span style={{ color: 'var(--td-light-pink)', fontWeight: 600, letterSpacing: 0, textTransform: 'none', marginLeft: 8 }}>{list.length} event{list.length === 1 ? '' : 's'} across {days.length} day{days.length === 1 ? '' : 's'}</span></FloatingTitle>
      <div style={{ display: 'flex', alignItems: 'center', gap: 20 }}>
        <Select label="Show" value={rep} options={['All', 'Regular', 'One-off']} onChange={v => { setRep(v); setShown(4); }} />
        <FilterToggle label={date ? 'From ' + fmt(date) : 'Pick a date'} open={pick} onToggle={() => setPick(o => !o)} style={{ margin: 0, padding: '0.5rem 0' }} />
      </div>
    </div>
    {pick && <div style={{ display: 'flex', justifyContent: 'flex-end', padding: '0 0 12px' }}><DatePicker value={date} onChange={v => { setDate(v); setPick(false); setShown(4); }} /></div>}
    <div className="td-grid2" style={{ paddingTop: 12 }}>{visible.flatMap(g => g.items).map(e => <EventCard key={e.key} day={String(e.dateObj.getDate()).padStart(2, '0')} month={MONTHS[e.dateObj.getMonth()]} title={e.title} time={e.time} repeat={e.repeat} location={e.venue} area={e.hood} onClick={() => navigate('/events/' + e.id)} />)}</div>
    {!days.length && <EmptyState>Nothing matches — try another filter.</EmptyState>}
    <div style={{ display: 'flex', gap: 28, justifyContent: 'center', marginTop: 44, flexWrap: 'wrap' }}>
      {more > 0 && <Button variant="pink" size="sm" onClick={() => setShown(s => s + 4)}>Show {Math.min(more, 4)} more day{Math.min(more, 4) === 1 ? '' : 's'}</Button>}
      <Button variant="dark" size="sm" href="#">Subscribe with iCal</Button><Button variant="dark" size="sm" href="#">Download as CSV</Button>
    </div>
  </div>;
}
