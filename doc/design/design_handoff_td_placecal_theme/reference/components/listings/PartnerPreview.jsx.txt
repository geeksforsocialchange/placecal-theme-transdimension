import React, { useState } from 'react';
import { Badge } from '../core/Badge.jsx';
/** Partners.elm entry: italic name with area badge on the right over a pink rule, summary beneath. */
export function PartnerPreview({ name, href = '#', area, summary, onClick, compact = false, style }) {
  const [h, setH] = useState(false);
  return <article onMouseEnter={() => setH(true)} onMouseLeave={() => setH(false)} style={{ margin: compact ? '20px 0 0' : '44px 0 0', fontFamily: 'var(--font-sans)', minWidth: 0, ...style }}>
    <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: '0.75rem', padding: '0.75rem 0', borderBottom: '2px solid ' + (h ? 'var(--td-white)' : 'var(--td-pink)'), transition: 'border 500ms linear' }}>
      <h3 style={{ margin: 0, fontSize: compact ? '1.1rem' : '1.25rem', fontStyle: 'italic', fontWeight: 500, lineHeight: 1.25, minWidth: 0, overflowWrap: 'anywhere' }}>
        <a href={href} onClick={e => { if (onClick) { e.preventDefault(); onClick(); } }} style={{ color: h ? 'var(--td-pink)' : 'var(--td-white)', textDecoration: 'none', transition: 'color 500ms linear' }}>{name}</a>
      </h3>
      <span style={{ display: 'flex', alignItems: 'center', gap: '0.5rem', flex: 'none' }}>{area && <Badge>{area}</Badge>}</span>
    </div>
    {summary && <p style={{ margin: '0 1rem 0 0', padding: '0.75rem 0', fontSize: compact ? '0.8777rem' : '1rem', lineHeight: 1.6, color: 'var(--td-white)' }}>{summary}</p>}
  </article>;
}
