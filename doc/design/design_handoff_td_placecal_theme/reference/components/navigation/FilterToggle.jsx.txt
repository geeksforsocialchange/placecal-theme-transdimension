import React from 'react';
/** "FILTER EVENTS +" — centred allcaps toggle that reveals PlaceCal's filter controls (XD). */
export function FilterToggle({ open = false, onToggle, label = 'Filter events', style }) {
  return <button onClick={onToggle} aria-expanded={open} style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', gap: '0.5rem', margin: '0 auto', background: 'none', border: 0, cursor: 'pointer', color: 'var(--td-white)', fontFamily: 'var(--font-sans)', fontWeight: 700, fontSize: '0.9rem', letterSpacing: '1.9px', textTransform: 'uppercase', padding: '0.5rem 1rem', ...style }}>
    {label}<span aria-hidden="true" style={{ color: 'var(--td-pink)', fontSize: '1.6rem', lineHeight: 0.8, fontWeight: 400 }}>{open ? '\u2212' : '+'}</span>
  </button>;
}
