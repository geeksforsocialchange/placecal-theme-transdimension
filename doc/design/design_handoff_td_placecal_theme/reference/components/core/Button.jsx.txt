import React, { useState } from 'react';
const V = {
  white: { bg: 'var(--td-white)', fg: 'var(--td-dark-blue)', bd: 'var(--td-white)', hbg: 'var(--td-purple)', hfg: 'var(--td-white)', hbd: 'var(--td-white)' },
  pink: { bg: 'var(--td-pink)', fg: 'var(--td-dark-blue)', bd: 'var(--td-pink)', hbg: 'var(--td-light-pink)', hfg: 'var(--td-dark-blue)', hbd: 'var(--td-light-pink)' },
  dark: { bg: 'var(--td-dark-blue)', fg: 'var(--td-white)', bd: 'var(--td-pink)', hbg: 'var(--td-purple)', hfg: 'var(--td-white)', hbd: 'var(--td-white)' },
  pinkOnLight: { bg: 'var(--td-pink)', fg: 'var(--td-dark-blue)', bd: 'var(--td-pink)', hbg: 'var(--td-purple)', hfg: 'var(--td-white)', hbd: 'var(--td-white)' },
  outline: { bg: 'transparent', fg: 'var(--td-white)', bd: 'var(--td-white)', hbg: 'var(--td-purple)', hfg: 'var(--td-white)', hbd: 'var(--td-white)' },
};
/** TD button: 0.3rem radius, 0.2rem border, 600 weight, colour-only 300ms hover. */
export function Button({ variant = 'pink', size = 'md', href, onClick, children, disabled, block, style, ...rest }) {
  const [h, setH] = useState(false);
  const v = V[variant] || V.pink;
  const sizes = { sm: { fontSize: '0.875rem', padding: '0.2rem 0.9rem 0.3rem', borderWidth: '2px' }, md: { fontSize: 'var(--button-font-size, 1.2rem)', padding: 'var(--button-padding, 0.375rem 1.25rem 0.5rem)', borderWidth: 'var(--button-border, 0.2rem)' } };
  const s = {
    display: block ? 'block' : 'inline-block', boxSizing: 'border-box', fontFamily: 'var(--font-sans)', fontWeight: 600, lineHeight: 1.5,
    textAlign: 'center', textDecoration: 'none', cursor: disabled ? 'default' : 'pointer', borderStyle: 'solid', borderRadius: 'var(--radius-sm, 0.3rem)',
    background: h && !disabled ? v.hbg : v.bg, color: h && !disabled ? v.hfg : v.fg, borderColor: h && !disabled ? v.hbd : v.bd,
    opacity: disabled ? 0.5 : 1, transition: 'background-color 300ms, border-color 300ms, color 300ms', ...sizes[size], ...style,
  };
  const Tag = href ? 'a' : 'button';
  return <Tag href={href} onClick={disabled ? undefined : onClick} style={s} onMouseEnter={() => setH(true)} onMouseLeave={() => setH(false)} disabled={!href && disabled} {...rest}>{children}</Tag>;
}
